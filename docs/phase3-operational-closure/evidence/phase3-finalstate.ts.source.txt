import 'dotenv/config';
import {PrismaClient} from '@prisma/client';
import {readFileSync,writeFileSync} from 'node:fs';
import {execFileSync} from 'node:child_process';
import {createHash} from 'node:crypto';
import {S3EvidenceStorage,storageConfig} from '../server/storage/s3';
import {ListObjectsV2Command,HeadObjectCommand} from '@aws-sdk/client-s3';
const root='docs/phase3-operational-closure/evidence',before=JSON.parse(readFileSync(root+'/storage-inventory.json','utf8')),baseline=JSON.parse(readFileSync(root+'/baseline.json','utf8'));
const u=new URL(process.env.DATABASE_URL!);if(u.hostname!=='127.0.0.1'||u.port!=='55432'||u.pathname!=='/communicate_pos')throw Error('READ_ONLY_LOCAL_REQUIRED');
const storage=new S3EvidenceStorage(storageConfig());if(new URL(storage.config.endpoint!).port!=='59000')throw Error('READ_ONLY_STORAGE_REQUIRED');const db=new PrismaClient();
try{const state=await db.$transaction(async tx=>{await tx.$executeRawUnsafe('SET TRANSACTION ISOLATION LEVEL REPEATABLE READ, READ ONLY');return {assets:await tx.evidenceAsset.findMany({select:{id:true,storageKey:true,status:true,fileSize:true,updatedAt:true}}),links:await tx.registrationEvidence.count(),documents:await tx.clientDocument.count(),counts:{clients:await tx.client.count(),equipment:await tx.equipment.count(),imeis:await tx.imei.count(),sales:await tx.sale.count(),requests:await tx.registrationRequest.count()}};});
const object=await storage.client.send(new ListObjectsV2Command({Bucket:storage.config.bucket}));const comparisons=[];for(const a of state.assets){const fingerprint=createHash('sha256').update(a.id).digest('hex').slice(0,16),old=before.assets.find((x:{idFingerprint:string})=>x.idFingerprint===fingerprint),head=await storage.client.send(new HeadObjectCommand({Bucket:storage.config.bucket,Key:a.storageKey}));comparisons.push({fingerprint,initialListedLastModified:old?.objectLastModified,finalListedLastModified:object.Contents?.find(o=>o.Key===a.storageKey)?.LastModified?.toISOString(),headLastModified:head.LastModified?.toISOString(),sameAssetState:!!old&&old.status===a.status&&old.size===a.fileSize&&old.updatedAt===a.updatedAt.toISOString(),samePhysicalMetadata:!!old&&old.objectLastModified===object.Contents?.find(o=>o.Key===a.storageKey)?.LastModified?.toISOString()&&head.ContentLength===old.size});}
const containers=baseline.containers.filter((c:{name:string})=>['communicate-pos-minio-1','communicate-pos-postgres-1'].includes(c.name)).map((old:{name:string;createdAt:string;startedAt:string;mounts:unknown})=>{const c=JSON.parse(execFileSync('docker',['inspect',old.name],{encoding:'utf8',windowsHide:true}))[0];return {name:old.name,status:c.State.Status,sameCreatedAt:old.createdAt===c.Created,sameStartedAt:old.startedAt===c.State.StartedAt,sameMounts:JSON.stringify(old.mounts)===JSON.stringify(c.Mounts.map((m:{Type:string;Name:string;Destination:string;RW:boolean})=>({type:m.Type,name:m.Name,destination:m.Destination,readWrite:m.RW})))};});
const unchanged=JSON.stringify(state.counts)===JSON.stringify(before.commercialCounts)&&state.links===before.postgresql.RegistrationEvidence&&state.documents===before.postgresql.ClientDocument&&object.Contents?.length===before.buckets.find((b:{configured:boolean})=>b.configured).objects&&comparisons.length===before.assets.length&&comparisons.every(c=>c.sameAssetState&&c.samePhysicalMetadata)&&containers.every(c=>c.sameCreatedAt&&c.sameStartedAt&&c.sameMounts&&c.status==='running');
const result={checkedAt:new Date().toISOString(),mode:'READ ONLY + LIST/HEAD; no further commercial photo download',commercialApplicationWrites:0,objects:object.Contents?.length,counts:state.counts,links:state.links,documents:state.documents,comparisons,habitualContainers:containers,unchangedDuringPhase3:unchanged,limits:'This checks aggregate counts and asset/object metadata, not a complete row-by-row commercial hash.'};writeFileSync(root+'/final-state.json',JSON.stringify(result,null,2));console.log(JSON.stringify({unchangedDuringPhase3:unchanged,objects:result.objects,habitualContainers:containers}));if(!unchanged)process.exitCode=1;
}finally{storage.client.destroy();await db.$disconnect();}
