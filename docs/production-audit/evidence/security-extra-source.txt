import {config} from 'dotenv';config({path:'.local/audit-isolated.env',override:true,quiet:true});config({path:'.local/audit-browser.env',override:true,quiet:true});
import {PrismaClient} from '@prisma/client';import request from 'supertest';import {readFileSync,writeFileSync} from 'node:fs';import {createApp} from '../server/app.js';import {S3EvidenceStorage,storageConfig} from '../server/storage/s3.js';
const state=JSON.parse(readFileSync('.local/audit-browser-state.json','utf8')),db=new PrismaClient(),storage=new S3EvidenceStorage(storageConfig()),app=createApp(db,{evidenceStorage:storage}),agent=request.agent(app),origin='http://127.0.0.1:5176';const results:any[]=[];const check=(name:string,pass:boolean,evidence:unknown)=>results.push({name,status:pass?'PASS':'FAIL',evidence});
try{
 await agent.post('/api/auth/login').set('Origin',origin).send({email:state.email,password:state.password});
 const doc=await db.clientDocument.findFirstOrThrow(),other=await db.client.findFirstOrThrow({where:{id:{not:doc.clientId}}});const foreign=await agent.get(`/api/clients/${other.id}/documents/${doc.id}/access`);check('foreign-client-document-scope',foreign.status===404,{status:foreign.status,signedUrlReturned:!!foreign.body.url});
 const malformed=await agent.post(`/api/clients/${doc.clientId}/documents`).query({purpose:'SYNTHETIC AUDIT',authorizationReference:'SYNTHETIC CONSENT',confirmed:'true',category:'DNI_FRONT',documentType:'MANUAL'}).set('Origin',origin).set('Content-Type','image/png').send(Buffer.from('not a photograph'));check('mime-spoofed-upload-rejected',malformed.status===400,{status:malformed.status,errorCode:malformed.body.error?.code});
 const sale=await db.sale.findUniqueOrThrow({where:{id:state.saleId}}),input={expectedUpdatedAt:sale.updatedAt.toISOString(),paymentMethod:'YAPE'};const updates=await Promise.all([agent.patch(`/api/sales/${sale.id}`).set('Origin',origin).send(input),agent.patch(`/api/sales/${sale.id}`).set('Origin',origin).send({...input,paymentMethod:'PLIN'})]);check('sale-edit-optimistic-concurrency',updates.map(r=>r.status).sort().join(',')==='200,409',{statuses:updates.map(r=>r.status)});
 const reports=await agent.get('/api/reports/general?preset=all');check('report-revenue-exact',reports.body.summary?.salesRevenue==='120.25',{status:reports.status,revenue:reports.body.summary?.salesRevenue});
 const seller=await db.user.findUniqueOrThrow({where:{email:'audit-seller@example.test'}}),limited=request.agent(app);await limited.post('/api/auth/login').set('Origin',origin).send({email:seller.email,password:state.password});await agent.post(`/api/users/${seller.id}/disable`).set('Origin',origin).send({});const denied=await limited.get('/api/auth/me');check('disabled-user-session-revoked',denied.status===401,{status:denied.status});
 const admin=await db.user.findUniqueOrThrow({where:{email:state.email}}),last=await agent.post(`/api/users/${admin.id}/disable`).set('Origin',origin).send({});check('last-superadmin-protected',last.status===409,{status:last.status,errorCode:last.body.error?.code});
 console.log(JSON.stringify(results));writeFileSync('docs/production-audit/evidence/additional-security-probes.json',JSON.stringify(results,null,2));
}finally{storage.client.destroy();await db.$disconnect();}
